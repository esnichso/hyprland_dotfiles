# AUR packages — installed with paru (preinstalled on CachyOS).

# GTK side of the Catppuccin Mocha palette.
catppuccin-gtk-theme-mocha

# Required by most GTK2/3 themes. Missing it silently breaks theming.
gtk-engine-murrine

# Cursor theme that stays sharp at 1x and 2x scale.
#
# The -bin variant on purpose. Plain `bibata-cursor-theme` renders the cursors
# from source at build time via python-clickgen, which pulls in Pillow, numpy
# and BLAS/LAPACK — roughly 20 packages, permanently installed, for a cursor
# theme. -bin is the same 2.0.7 release, prebuilt, with no dependencies at all.
bibata-cursor-theme-bin

# The other Catppuccin flavours, for themes/frappe.toml and themes/macchiato.toml.
# Skip these if you only ever use Mocha — the themes still work without them,
# they just fall back to a recoloured Adwaita (see themes/tokyo-night.toml).
catppuccin-gtk-theme-frappe
catppuccin-gtk-theme-macchiato
catppuccin-gtk-theme-latte
